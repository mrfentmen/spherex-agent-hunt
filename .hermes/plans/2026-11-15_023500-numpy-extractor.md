```python
"""FITS HDU1 -> detections CSV. numpy only. No astropy."""
from __future__ import annotations
import numpy as np
def background_mad(img: np.ndarray) -> tuple[float, float]:
    med = float(np.median(img))
    mad = float(np.median(np.abs(img - med)))
    return med, 1.4826 * mad
def find_peaks(img: np.ndarray, nsigma: float = 5.0) -> list[tuple[int, int, float]]:
    med, sigma = background_mad(img)
    thresh = med + nsigma * sigma
    ys, xs = np.where(img > thresh)
    # local-maximum filter, 3x3
    out = []
    for y, x in zip(ys.tolist(), xs.tolist()):
        y0, y1 = max(0, y-1), min(img.shape[0], y+2)
        x0, x1 = max(0, x-1), min(img.shape[1], x+2)
        if img[y, x] == img[y0:y1, x0:x1].max():
            out.append((x, y, float(img[y, x])))
    # shortcut: 8-connected dedupe is approximate, upgrade to label() when false splits appear.
    return out
```

import struct
def read_hdu1(path: str):
    """Return (header dict, image float32 2D array). HDU0 skipped, HDU1 read."""
    with open(path, 'rb') as f:
        for hdu in range(2):
            cards = []
            while True:
                blk = f.read(2880)
                if not blk:
                    raise ValueError('EOF in header')
                for i in range(0, 2880, 80):
                    cards.append(blk[i:i+80].decode('ascii', 'replace'))
                if any(c.startswith('END') for c in cards[-36:]):
                    break
            hdr = {c[:8].strip(): c[10:30].strip() for c in cards if '=' in c[:10]}
            naxis = int(hdr.get('NAXIS', '0').split('/')[0])
            naxes = [int(hdr.get(f'NAXIS{n}', '0').split('/')[0]) for n in range(1, naxis+1)]
            import math
            nbytes = math.ceil(abs(int(hdr.get('BITPIX', '8').split('/')[0]))//8 * (math.prod(naxes) if naxes and all(naxes) else 0) / 2880) * 2880 if naxis else 0
            if hdu == 0:
                if nbytes:
                    f.seek(nbytes, 1)
                continue
            import numpy as np
            raw = f.read(nbytes) if nbytes else b''
            n1, n2 = naxes[0], naxes[1]
            arr = np.frombuffer(raw[:n1*n2*4], dtype='>f4').reshape((n2, n1)).astype(np.float32)
            return hdr, arr
    raise ValueError('no HDU1')
```

def moments(img, x, y, r=4):
    import numpy as np
    y0, y1 = max(0, y-r), min(img.shape[0], y+r+1)
    x0, x1 = max(0, x-r), min(img.shape[1], x+r+1)
    cut = img[y0:y1, x0:x1].astype(float)
    cut -= cut.min()
    s = cut.sum() or 1.0
    gy, gx = np.mgrid[y0:y1, x0:x1]
    cx, cy = (gx*cut).sum()/s, (gy*cut).sum()/s
    muxx = ((gx-cx)**2*cut).sum()/s; myyy = ((gy-cy)**2*cut).sum()/s
    area = float((cut > cut.max()*0.5).sum())
    rnd = abs(muxx-myyy)/(muxx+myyy+1e-9)
    return cx, cy, rnd, area, float(s)
```

## Risks, tradeoffs, and open questions
- Linear TAN vs SIP: 3-arcsec linking tolerance may absorb error near field center but fail at edges; fallback is full SIP inversion or astropy if allowed.
- Numpy peaks without deblend + threshold tuning will false-positive on bright-star wings; mitigate with VARIANCE/FLAGS masking, raise nsigma, cap to brightest K per frame.
- Same-MJD multi-detector frames are one epoch; enforce distinct-MJD selection or hunt returns 0 tracks and wastes 130+ MB downloads.
- 65 MB x 3 = ~195 MB over venue wifi; keep 0.05 deg cones, HEAD first, delete FITS immediately.
- Open: which 3 MJDs? Pick from `/tmp/inv_3ep.csv` with >= 1 day gaps. Open: SNR definition? Use `(peak-med)/sigma` now, switch to variance-pixel SNR when VARIANCE HDU reader lands.
